 @classmethod
 def cleanup(cls):
  if cls.started:
   status=cls.runpg('pg_ctl',['-D',str(cls.data),'status']).returncode
   if status==0:cls.runpg('pg_ctl',['-D',str(cls.data),'-m','fast','-w','stop']).check_returncode()
   elif status!=3:raise RuntimeError('OWNED_PG_STOP_UNKNOWN_RETAINED')
   assert cls.runpg('pg_ctl',['-D',str(cls.data),'status']).returncode==3
  shutil.rmtree(cls.root)
 def setUp(self):
  self.sql("DROP SCHEMA public CASCADE;CREATE SCHEMA public;GRANT USAGE ON SCHEMA public TO PUBLIC;DO $$BEGIN IF NOT EXISTS(SELECT 1 FROM pg_roles WHERE rolname='did_reader')THEN CREATE ROLE did_reader NOLOGIN;END IF;END$$;").check_returncode()
  for table,columns in self.columns.items():
   self.sql('CREATE TABLE public.'+table+'('+','.join('"'+c+'" text'for c in [*columns,'synthetic_ungranted'])+');GRANT SELECT('+','.join('"'+c+'"'for c in columns)+')ON public.'+table+' TO did_reader;').check_returncode()
 def gate(self):
  result=self.sql('SET ROLE did_reader;PREPARE gate(jsonb)AS '+self.query+";EXECUTE gate('"+json.dumps(self.columns).replace("'","''")+"'::jsonb);")
  self.assertEqual(result.returncode,0,result.stderr);return result.stdout.strip().split('|')
 def test_exact_eight_relation_select_scope_is_accepted(self):self.assertEqual(self.gate(),['t']*6)
 def test_extra_select_or_dml_or_sequence_authority_is_refused(self):
  for grant,revoke,index in [('GRANT SELECT(synthetic_ungranted)ON tenants TO did_reader','REVOKE SELECT(synthetic_ungranted)ON tenants FROM did_reader',2),('GRANT UPDATE(status)ON tenants TO did_reader','REVOKE UPDATE(status)ON tenants FROM did_reader',2),('CREATE SEQUENCE did_forbidden;GRANT USAGE ON SEQUENCE did_forbidden TO did_reader','DROP SEQUENCE did_forbidden',4)]:
   with self.subTest(grant=grant):self.sql(grant+';').check_returncode();self.assertEqual(self.gate()[index],'f');self.sql(revoke+';').check_returncode()
